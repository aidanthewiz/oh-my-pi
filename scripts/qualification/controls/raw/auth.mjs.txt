function observeStore(store) {
  let closed = false;
  const calls = [], restores = [];
  for (const method of ['listAuthCredentials', 'tryUpdateAuthCredentialIfMatches', 'getCache', 'setCache', 'upsertCredentialBlock', 'getCredentialBlock', 'close']) {
    const original = store[method];
    if (typeof original !== 'function') continue;
    const descriptor = Object.getOwnPropertyDescriptor(store, method);
    Object.defineProperty(store, method, { configurable: true, writable: true, value: function (...args) {
      const row = { method, afterClose: closed };
      calls.push(row);
      try {
        const value = Reflect.apply(original, this, args);
        if (method === 'close') closed = true;
        return value;
      } catch (error) { row.error = String(error); throw error; }
    } });
    restores.push(() => descriptor ? Object.defineProperty(store, method, descriptor) : Reflect.deleteProperty(store, method));
  }
  return { calls, restore() { for (const fn of restores.reverse()) fn(); } };
}
