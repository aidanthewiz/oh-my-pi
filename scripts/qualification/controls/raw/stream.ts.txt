async function bounded<T>(promise: Promise<T>, label: string): Promise<T> {
	let timer: NodeJS.Timeout | undefined;
	try {
		return await Promise.race([promise, new Promise<never>((_, reject) => {
			timer = setTimeout(() => reject(new Error(`deadline: ${label}`)), 8000);
		})]);
	} finally {
		clearTimeout(timer);
	}
}
interface Journal<T> {
	values: T[];
	push(value: T): void;
	wait(predicate: (value: T) => boolean): Promise<T>;
}
function journal<T>(): Journal<T> {
	const values: T[] = [];
	const waiters: Array<{ predicate: (value: T) => boolean; resolve: (value: T) => void }> = [];
	return {
		values,
		push(value: T) {
			values.push(value);
			for (let i = waiters.length - 1; i >= 0; i--) {
				if (!waiters[i]!.predicate(value)) continue;
				waiters.splice(i, 1)[0]!.resolve(value);
			}
		},
		wait(predicate: (value: T) => boolean): Promise<T> {
			const found = values.find(predicate);
			if (found !== undefined) return Promise.resolve(found);
			const pending = Promise.withResolvers<T>();
			waiters.push({ predicate, resolve: pending.resolve });
			return bounded(pending.promise, "observed payload");
		},
	};
}
