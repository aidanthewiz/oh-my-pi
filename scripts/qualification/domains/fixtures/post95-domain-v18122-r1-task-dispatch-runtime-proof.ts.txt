function stream(model: Model<Api>, context: Context): AssistantMessageEventStream {
 assert.equal(model.provider, provider); assert.equal(model.id, "owned"); assert.equal(model.api, api);
 const tools = context.tools ?? [];
 const isChild = tools.some(tool => tool.name === "read" || tool.name === "yield");
 const structured = tools.some(tool => tool.name === "yield");
 let content: AssistantMessage["content"];
 let stopReason: "stop" | "toolUse" = "stop";
 if (!isChild) content = [{ type: "text", text: "Owned local task label" }];
 else if (scenario === "empty") content = [];
 else if (scenario === "structured" && context.messages.at(-1)?.role !== "toolResult") {
  assert(structured, "Structured child did not receive yield");
  content = [{ type: "toolCall", id: `yield-${randomUUID()}`, name: "yield", arguments: { type: "result", data: { answer: 42 } } }];
  stopReason = "toolUse";
 } else if (scenario === "structured") content = [{ type: "text", text: "Accepted result submitted" }];
 else if (scenario === "chatter") content = [{ type: "text", text: `Peer acknowledgment ${marker}` }];
 else {
  assert(!structured, "Default plain child unexpectedly received yield");
  content = [{ type: "text", text: scenario === "manual" ? manualReport : `  Initial plain report ${marker}\nexact final whitespace.  ` }];
 }
 const message: AssistantMessage = { role: "assistant", api: model.api, provider: model.provider, model: model.id,
  content, stopReason, usage: structuredClone(usage), timestamp: Date.now() };
 const observation = { scenario, model: `${model.provider}/${model.id}`, tools: JSON.parse(JSON.stringify(tools)),
  messages: JSON.parse(JSON.stringify(context.messages)), content, stopReason, streamEvents: ["start"] };
 wire.push(observation);
 const events = new AssistantMessageEventStream();
 events.push({ type: "start", partial: { ...message, content: [] } });
 for (const [contentIndex, block] of content.entries()) {
  const partial = { ...message, content: content.slice(0, contentIndex + 1) };
  if (block.type === "toolCall") {
   events.push({ type: "toolcall_start", contentIndex, partial });
   events.push({ type: "toolcall_delta", contentIndex, delta: JSON.stringify(block.arguments), partial });
   events.push({ type: "toolcall_end", contentIndex, toolCall: block, partial });
   observation.streamEvents.push("toolcall_start", "toolcall_delta", "toolcall_end");
  } else if (block.type === "text") {
   events.push({ type: "text_start", contentIndex, partial });
   events.push({ type: "text_delta", contentIndex, delta: block.text, partial });
   events.push({ type: "text_end", contentIndex, content: block.text, partial });
   observation.streamEvents.push("text_start", "text_delta", "text_end");
  }
 }
 events.push({ type: "done", reason: stopReason, message }); events.end(message);
 observation.streamEvents.push("done");
 return events;
}
