// This adapter records production ANSI/TSP bytes and applies frames with the production wire consumer.
class WireConsumer implements Terminal {
	readonly columns = 108;
	readonly rows = 40;
	readonly kittyProtocolActive = false;
	readonly kittyEnableSequence = null;
	readonly appearance = "dark" as const;
	readonly docs = new Map<string, InstanceType<typeof TspDocument>>();
	readonly frames: TspFrame[] = [];
	readonly errors: unknown[] = [];
	readonly wire: { verb: string; body: unknown }[] = [];
	readonly chunks = new Map<string, string>();
	readonly blobs = new Set<string>();
	bytes = "";
	surface: string | undefined;
	input: ((data: string) => void) | undefined;
	started = false;
	constructor(readonly media: boolean, readonly nativeMode = true) {}
	start(input: (data: string) => void): void { this.input = input; this.started = true; }
	stop(): void { this.input = undefined; this.started = false; }
	async drainInput(): Promise<void> { await Bun.sleep(0); }
	onTspHello(callback: TspHelloHandler): void {
		queueMicrotask(() => {
			if (this.started) callback(this.nativeMode ? { r: "hello", v: 1, term: "coreforge-owned-proof",
				kinds: this.media ? TSP_KINDS : TSP_KINDS.filter(k => k !== "media"), features: ["blobs", "settle", "dock"],
				cols: this.columns, dark: true, reduceMotion: true, hour12: false } : null);
		});
	}
	send(event: TspEvent): void { assert(this.input); this.input(`\x1b_tsp;e;${JSON.stringify(event)}\x1b\\`); }
	write(bytes: string): void {
		this.bytes += bytes;
		let at = 0;
		for (;;) {
			const start = bytes.indexOf("\x1b_tsp;", at);
			if (start < 0) break;
			const end = bytes.indexOf("\x1b\\", start);
			assert(end >= 0);
			const message = splitTspMessage(bytes.slice(start, end + 2)); assert(message);
			at = end + 2;
			let body = message.body;
			if (message.params.c !== undefined) {
				const id = message.params.c;
				body = (this.chunks.get(id) ?? "") + body;
				if (message.params.m === "1") { this.chunks.set(id, body); continue; }
				this.chunks.delete(id);
			}
			if (message.verb === "b") { this.blobs.add(digest(Buffer.from(body, "base64"))); continue; }
			const data = JSON.parse(body); this.wire.push({ verb: message.verb, body: data });
			if (message.verb === "q" && data.q === "blobs") {
				queueMicrotask(() => this.input?.(`\x1b_tsp;r;${JSON.stringify({ r: "blobs", have: data.ids.filter((id: string) => this.blobs.has(id)) })}\x1b\\`));
			} else if (message.verb === "o") {
				this.docs.set(data.id, new TspDocument(data.id)); this.surface = data.id;
			} else if (message.verb === "f") {
				this.frames.push(data);
				const doc = this.docs.get(data.sf); assert(doc); this.errors.push(...doc.applyFrame(data));
				queueMicrotask(() => this.input && this.send({ ev: "ack", sf: data.sf, s: data.s }));
			} else if (message.verb === "x") {
				if (data.keep) this.docs.get(data.id)?.close(); else this.docs.delete(data.id);
			}
		}
	}
	moveBy(n: number): void { this.write(n < 0 ? `\x1b[${-n}A` : `\x1b[${n}B`); }
	hideCursor(): void { this.write("\x1b[?25l"); }
	showCursor(): void { this.write("\x1b[?25h"); }
	clearLine(): void { this.write("\x1b[2K"); }
	clearFromCursor(): void { this.write("\x1b[J"); }
	clearScreen(): void { this.write("\x1b[2J\x1b[H"); }
	setTitle(title: string): void { this.write(`\x1b]2;${title}\x07`); }
	setProgress(active: boolean): void { this.write(`\x1b]9;4;${active ? 3 : 0};\x07`); }
	onAppearanceChange(callback: (appearance: "dark" | "light") => void): void { callback("dark"); }
	find(predicate: (node: TspNode) => boolean): TspNode[] {
		const doc = this.surface && this.docs.get(this.surface); assert(doc);
		const out: TspNode[] = [];
		const visit = (node: TspNode): void => { if (predicate(node)) out.push(node); for (const child of node.c ?? []) visit(child); };
		visit(doc.snapshot()); return out;
	}
}
