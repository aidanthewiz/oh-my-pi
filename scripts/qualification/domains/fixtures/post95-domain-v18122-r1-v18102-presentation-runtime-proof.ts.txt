const sha = (bytes: Uint8Array | string) => createHash("sha256").update(bytes).digest("hex");
const required = (name: string) => {

	assert.ok(value, `${name} is required`);
	return value;
};
const walk = (root: TspNode): TspNode[] => [root, ...(root.c ?? []).flatMap(walk)];

// This private terminal is a protocol peer, not a host desktop or user terminal.
class PrivateTerminal implements Terminal {
	readonly columns = 150;
	readonly rows = 40;
	readonly kittyProtocolActive = false;
	readonly kittyEnableSequence = null;
	readonly appearance = "dark" as const;
	readonly docs = new Map<string, TspDocument>();
	readonly errors: unknown[] = [];
	readonly chunks = new Map<string, string>();
	readonly blobs = new Map<string, Uint8Array>();
	readonly writes: string[] = [];
	input: ((data: string) => void) | undefined;
	hello: TspHello | null | undefined;
	callbacks: TspHelloHandler[] = [];
	constructor(readonly images: boolean) {}
	start(input: (data: string) => void) {
		this.input = input;
		queueMicrotask(() => {
			this.hello = { r: "hello", v: 1, term: "coreforge-private-proof", kinds: TSP_KINDS.filter(kind => this.images || kind !== "image"), features: [], cols: this.columns, dark: true, reduceMotion: true };
			for (const callback of this.callbacks) callback(this.hello);
		});
	}
	onTspHello(callback: TspHelloHandler) { this.callbacks.push(callback); if (this.hello !== undefined) callback(this.hello); }
	stop() { this.input = undefined; this.callbacks.length = 0; }
	async drainInput() { await Promise.resolve(); }
	send(data: string) { this.input?.(data); }
	write(data: string) {
		this.writes.push(data);
		for (const sequence of data.matchAll(/\x1b_tsp;[\s\S]*?\x1b\\/g)) {
			const raw = splitTspMessage(sequence[0]);
			assert.ok(raw, "valid production TSP encoding");
			let body = raw.body;
			if (raw.params.c !== undefined) {
				body = (this.chunks.get(raw.params.c) ?? "") + body;
				if (raw.params.m === "1") { this.chunks.set(raw.params.c, body); continue; }
				this.chunks.delete(raw.params.c);
			}
			if (raw.verb === "o") { const open = JSON.parse(body); this.docs.set(open.id, new TspDocument(open.id)); }
			if (raw.verb === "x") { const close = JSON.parse(body); this.docs.delete(close.id); }
			if (raw.verb === "b") {
				const bytes = Buffer.from(body, "base64");
				assert.equal(sha(bytes), raw.params.id);
				this.blobs.set(raw.params.id!, bytes);
			}
			if (raw.verb === "f") {
				const frame = JSON.parse(body) as TspFrame;
				const doc = this.docs.get(frame.sf);
				assert.ok(doc, "frame has an opened surface");
				this.errors.push(...doc.applyFrame(frame));
				queueMicrotask(() => this.send(`\x1b_tsp;e;${JSON.stringify({ ev: "ack", sf: frame.sf, s: frame.s })}\x1b\\`));
			}
		}
	}
	nodes() { return [...this.docs.values()].flatMap(doc => walk(doc.snapshot())); }
	moveBy(lines: number) { this.write(`\x1b[${Math.abs(lines)}${lines < 0 ? "A" : "B"}`); }
	hideCursor() { this.write("\x1b[?25l"); }
	showCursor() { this.write("\x1b[?25h"); }
	clearLine() { this.write("\x1b[2K"); }
	clearFromCursor() { this.write("\x1b[J"); }
	clearScreen() { this.write("\x1b[2J\x1b[H"); }
	setTitle(title: string) { this.write(`\x1b]0;${title}\x07`); }
	setProgress(active: boolean) { this.write(`\x1b]9;4;${active ? 3 : 0}\x07`); }
	onAppearanceChange(callback: (appearance: "dark" | "light") => void) { callback("dark"); }
}

		record("private-git-native-and-fallback-image-binary", { image: { before: sha(red), after: sha(gray), refs: [image.oldRef, image.newRef], delivery: "owned-content-addressed-blob" }, binary: { before: sha(binaryOld), after: sha(binaryNew), refs: [binary.oldRef, binary.newRef], spoofedRasterExtension: true }, svg: { before: sha(svgOld), after: sha(svgNew), delivery: "rasterized-owned-blob" }, mutableSourceReplacement: "retained-delivered-bytes", hostMedia: false, capabilities: ["image-and-media", "media-without-image"], paidRequests: 0 });
